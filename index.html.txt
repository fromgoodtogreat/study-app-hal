        return {

      map[key].total += 1;
      if (q.isCorrect || q.isSolved) {
        map[key].correct += 1;

テスト種別: ${uploadForm.type}
科目は算数・国語・理科・社会を含めてください。`

                    questionSummary: { type: "STRING" }
                  }
                }
              }
            }
          }
        }
      };

        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      const data = await res.json();
      const parsed = JSON.parse(data?.candidates?.[0]?.content?.parts?.[0]?.text || "{}");

          questions: parsed.questions.map((q, idx) => ({

          }))
        }));
        showToast('アップロードされたファイルから問題と単元を読み取りました！', 15);
      } else {
        throw new Error('Empty');
      }
    } catch (e) {
      // オフライン・API未設定時のスマート入力補助
      setUploadForm(prev => ({
        ...prev,
        name: prev.name || `第1回 ${prev.type}`,
        questions: [
          { id: `scan-${Date.now()}-1`, subject: '算数', number: '大問2(2)', unit: '和差算', isCorrect: false, accuracyRate: 58, questionSummary: '2人の持っている 금액の和と差からそれぞれの金額を求める問題' },
          { id: `scan-${Date.now()}-2`, subject: '算数', number: '大問4(1)', unit: '植木算', isCorrect: false, accuracyRate: 42, questionSummary: '木と木の間の長さと本数から全体の距離を求める問題' },
          { id: `scan-${Date.now()}-3`, subject: '算数', number: '大問5(2)', unit: '角度', isCorrect: true, accuracyRate: 65, questionSummary: '平行線と折り返した図形の角度を求める問題' },
          { id: `scan-${Date.now()}-4`, subject: '理科', number: '大問3 問2', unit: '磁石の性質', isCorrect: false, accuracyRate: 35, questionSummary: '電磁石の導線の巻き数と方位磁針の振れ方の問題' }
        ]
      }));

      (Number(s.算数.score) || 0) +
      (Number(s.国語.score) || 0) +
      (Number(s.理科.score) || 0) +
      (Number(s.社会.score) || 0);

    const validDevs = ['算数', '国語', '理科', '社会']
      .map(k => Number(s[k].deviation))
      .filter(v => v > 0);
    const avgDev = validDevs.length > 0
      ? Number((validDevs.reduce((a, b) => a + b, 0) / validDevs.length).toFixed(1))
      : 50.0;

    const newTest = {
      id: `test-${Date.now()}`,
      name: uploadForm.name.trim() || `${uploadForm.type} (${uploadForm.date})`,
      type: uploadForm.type,
      date: uploadForm.date,
      course: uploadForm.course.trim() || '標準',
      files: uploadForm.files,
      subjects: {
        算数: { score: Number(s.算数.score) || 0, maxScore: Number(s.算数.maxScore), deviation: Number(s.算数.deviation) || 50 },
        国語: { score: Number(s.国語.score) || 0, maxScore: Number(s.国語.maxScore), deviation: Number(s.国語.deviation) || 50 },
        理科: { score: Number(s.理科.score) || 0, maxScore: Number(s.理科.maxScore), deviation: Number(s.理科.deviation) || 50 },

                        <div key={idx} className="py-3 flex flex-col sm:flex-row sm:items-center justify-between gap-2">
                          <span className="font-bold text-sm text-[#3D352E] sm:w-44 truncate">{u.unit}</span>
                          <div className="flex-1 bg-[#EFEAE0] h-3 rounded-full overflow-hidden mx-2">
                            <div
                              className={`h-full rounded-full transition-all ${isGood ? 'bg-[#4B8364]' : 'bg-[#C86D65]'}`}
                              style={{ width: `${Math.max(4, pct)}%` }}
                            />
                          </div>
                          <div className="flex items-center justify-end gap-4 sm:w-56 text-xs">
                            <span className="font-extrabold text-[#3D352E] w-12 text-right">{pct}%</span>
                            <span className="text-[#7A6F63] w-12">{u.correct}/{u.total}問</span>
                            <span className={`font-bold w-24 text-right ${u.missedHighAcc > 0 ? 'text-[#C86D65]' : 'text-[#4B8364]'}`}>
                              {u.missedHighAcc > 0 ? `取りこぼし ${u.missedHighAcc}` : '取りこぼし 0'}
                            </span>
                          </div>
                        </div>
                      );
                    })}
                </div>
              )}
            </div>

            {/* 単元マップ（左が土台、右へ行くほど発展のフロー表示 / image_0.png 再現） */}
            <div className="space-y-3">
              <p className="text-xs text-[#7A6F63] font-bold">
                左が土台、右へ行くほど発展。線は「この単元ができると、次の単元が解きやすくなる」つながりです。単元をタップするとAI演習ができます。
              </p>

              <div className="space-y-4">
                {CURRICULUM_MAP[selectedSubject].map((cat, idx) => (
                  <div key={idx} className="bg-[#EFEAE0]/70 border border-[#E2D9C8] rounded-3xl p-4 sm:p-5 space-y-3">
                    <h4 className="text-xs font-extrabold text-[#5C5248]">{cat.category}</h4>
                    <div className="flex flex-wrap items-center gap-3">
                      {cat.units.map((unitName, uIdx) => {
                        const stat = unitStatsMap[`${selectedSubject}:${unitName}`];
                        const pct = stat ? Math.round((stat.correct / stat.total) * 100) : null;
                        let borderColor = 'border-[#D5CCB8] bg-[#FAF8F4] text-[#7A6F63]';
                        if (pct !== null) {
                          if (pct >= 80) borderColor = 'border-[#4B8364] bg-[#E6EFE9] text-[#2C5840]';
                          else if (pct >= 50) borderColor = 'border-[#D49B35] bg-[#FDF3DC] text-[#805B10]';
                          else borderColor = 'border-[#C86D65] bg-[#F5E6E4] text-[#8F3A34]';
                        }

                        return (
                          <React.Fragment key={unitName}>
                            <div
                              onClick={() => {
                                const targetQ = stat?.questions?.[0] || {
                                  id: `map-${selectedSubject}-${unitName}`,
                                  subject: selectedSubject,
                                  number: '単元演習',
                                  unit: unitName,
                                  isCorrect: pct !== null && pct >= 80,
                                  accuracyRate: 55,
                                  questionSummary: `${unitName}の基本〜標準問題`
                                };
                                startDojoExercise(targetQ, pct !== null && pct >= 80 ? 'advanced' : 'review');
                              }}
                              className={`cursor-pointer border-2 rounded-2xl px-4 py-3 min-w-[140px] shadow-sm hover:scale-[1.02] transition ${borderColor}`}
                            >
                              <p className="font-extrabold text-xs sm:text-sm">{unitName}</p>
                              <p className="text-[11px] mt-1 font-bold opacity-80">
                                {stat ? `${pct}% (${stat.total}問)` : '未登録（タップで演習）'}
                              </p>
                            </div>
                            {uIdx < cat.units.length - 1 && (
                              <span className="text-[#B8AFA0] font-bold hidden sm:inline">―▶</span>
                            )}
                          </React.Fragment>
                        );
                      })}
                    </div>
                  </div>
                ))}
              </div>
            </div>
          </div>
        )}

        {/* ==================== 3. AI道場・演習画面（image_2.png の「算数の演習」UI忠実再現） ==================== */}
        {activeView === 'dojo' && (
          <div className="space-y-6">
            {/* 上部タイトルと正答率フィルター切り替え */}
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
              <div>
                <h2 className="text-2xl font-extrabold text-[#3D352E]">
                  AI道場（解き直し＆類題・応用演習）
                </h2>
                <p className="text-xs sm:text-sm text-[#7A6F63] mt-0.5">
                  間違えた問題は正答率基準（30% / 40% / 50%以上）で絞り込んで復習し、得意な問題は応用問題に挑戦できます。
                </p>
              </div>

              {/* 復習の正答率基準切り替えボタン */}
              <div className="flex flex-wrap gap-2 bg-[#FFFDF9] p-1.5 rounded-full border border-[#D8C5A5]">
                {[30, 40, 50].map(val => (
                  <button
                    key={val}
                    onClick={() => setAccuracyThreshold(val)}
                    className={`px-3.5 py-1.5 rounded-full text-xs font-extrabold transition ${
                      accuracyThreshold === val
                        ? 'bg-[#4B8364] text-white shadow-sm'
                        : 'text-[#5C5248] hover:bg-[#F3EFE6]'
                    }`}
                  >
                    正答率{val}%以上
                  </button>
                ))}
              </div>
            </div>

            {/* 演習中の問題カード（image_2.png 再現） */}
            {activeQuestion && (
              <div className="bg-[#FFFDF9] border-2 border-[#D8C5A5] rounded-3xl p-5 sm:p-7 shadow-md space-y-5">
                <div className="flex flex-wrap items-center justify-between gap-2 border-b border-[#EFEAE0] pb-3">
                  <div className="flex items-center gap-2">
                    <span
                      className={`px-3 py-1 rounded-full text-xs font-extrabold ${
                        activeQuestion.mode === 'review'
                          ? 'bg-[#F5E6E4] text-[#A64B44]'
                          : 'bg-[#E6EFE9] text-[#3B6E52]'
                      }`}
                    >
                      {activeQuestion.mode === 'review' ? 'まちがえた（解き直し＆類題）' : '得意を伸ばす（応用問題）'}
                    </span>
                    <span className="text-xs sm:text-sm font-bold text-[#7A6F63]">
                      {activeQuestion.subject}・{activeQuestion.unit} ・ もとの問題 {activeQuestion.number} (正答率{activeQuestion.accuracyRate}%)
                    </span>
                  </div>
                  <button
                    onClick={() => setActiveQuestion(null)}
                    className="text-xs font-bold text-[#5C5248] border border-[#D8C5A5] px-3 py-1 rounded-lg hover:bg-[#F3EFE6]"
                  >
                    問題一覧にもどる
                  </button>
                </div>

                {isGeneratingAi ? (
                  <div className="py-12 text-center space-y-3">
                    <RefreshCw className="w-8 h-8 text-[#4B8364] animate-spin mx-auto" />
                    <p className="text-sm font-bold text-[#5C5248]">
                      パーソナルコーチが「{activeQuestion.unit}」の解説と演習問題を作成しています...
                    </p>
                  </div>
                ) : (
                  aiGeneratedContent && (
                    <div className="space-y-5">
                      {/* 問題文 */}
                      <p className="text-base sm:text-lg font-bold text-[#3D352E] leading-relaxed bg-[#FAF8F4] p-4 rounded-2xl border border-[#EAE3D5]">
                        {aiGeneratedContent.problemText}
                      </p>

                      {/* こたえ入力欄 ＋ こたえあわせボタン（image_2.png 再現） */}

                        <input
                          type="text"
                          placeholder="こたえを入力（例: 50、3/4、ア など）またはノートに解いて確認"
                          value={userAnswerInput}
                          onChange={e => setUserAnswerInput(e.target.value)}
                          className="flex-1 bg-white border border-[#C9BFA8] rounded-2xl px-4 py-3 text-sm sm:text-base font-bold text-[#3D352E] focus:outline-none focus:border-[#4B8364]"
                        />
                        <button
                          type="button"
                          onClick={() => {
                            if (
                              userAnswerInput.trim() !== '' &&
                              aiGeneratedContent.expectedAnswer &&
                              userAnswerInput.includes(aiGeneratedContent.expectedAnswer.replace(/[^0-9a-zA-Zぁ-んァ-ヶ一-龠]/g, ''))
                            ) {
                              setAnswerFeedback('correct');
                            } else {
                              setAnswerFeedback('checked');
                            }
                            setShowFullExplanation(true);
                          }}
                          className="bg-[#4B8364] hover:bg-[#3B6E52] text-white font-extrabold px-6 py-3 rounded-2xl shadow-sm transition shrink-0"
                        >
                          こたえあわせ
                        </button>
                      </div>

                      {/* ヒント表示ボックス */}
                      {hintLevel >= 1 && (
                        <div className="bg-[#F5EFE0] border border-[#E2D3B3] rounded-2xl p-4 text-xs sm:text-sm text-[#4A3F31] font-medium leading-relaxed">
                          <strong>ヒント1：</strong> {aiGeneratedContent.hint1}
                        </div>
                      )}
                      {hintLevel >= 2 && (
                        <div className="bg-[#F5EFE0] border border-[#E2D3B3] rounded-2xl p-4 text-xs sm:text-sm text-[#4A3F31] font-medium leading-relaxed">
                          <strong>ヒント2：</strong> {aiGeneratedContent.hint2}
                        </div>
                      )}

                      {/* ヒントボタン ＆ こたえととき方を見るボタン（image_2.png 再現） */}
                      <div className="flex flex-wrap items-center gap-3">
                        {hintLevel < 2 && (
                          <button
                            type="button"
                            onClick={() => setHintLevel(prev => prev + 1)}
                            className="bg-white hover:bg-[#F3EFE6] text-[#2C5840] border-2 border-[#4B8364] font-extrabold text-xs sm:text-sm px-4 py-2 rounded-xl transition"
                          >
                            ヒント{hintLevel + 1}を見る
                          </button>
                        )}
                        <button
                          type="button"
                          onClick={() => setShowFullExplanation(!showFullExplanation)}
                          className="bg-white hover:bg-[#F3EFE6] text-[#5C5248] border border-[#D8C5A5] font-bold text-xs sm:text-sm px-4 py-2 rounded-xl transition"
                        >
                          {showFullExplanation ? '解説を閉じる' : 'こたえと とき方を見る'}
                        </button>
                        <button
                          type="button"
                          onClick={() => startDojoExercise(activeQuestion, activeQuestion.mode)}
                          className="text-xs font-bold text-[#4B8364] hover:underline ml-auto flex items-center gap-1"
                        >
                          <RefreshCw className="w-3.5 h-3.5" />
                          別の類題をつくる
                        </button>
                      </div>

                      {/* 詳しいステップ解説と答え */}
                      {showFullExplanation && (
                        <div className="bg-[#F4F8F5] border border-[#B8D4C4] rounded-2xl p-5 space-y-3">
                          <div className="flex items-center justify-between">
                            <p className="text-sm sm:text-base font-extrabold text-[#2C5840]">
                              ⭕ こたえ：{aiGeneratedContent.expectedAnswer}
                            </p>
                            {answerFeedback === 'correct' && (
                              <span className="bg-[#4B8364] text-white text-xs font-bold px-3 py-1 rounded-full">
                                大正解！すばらしい！
                              </span>
                            )}
                          </div>

                          <div className="space-y-2 pt-2">
                            <p className="text-xs font-extrabold text-[#3B6E52]">【くわしい解き方・ステップ解説】</p>
                            {aiGeneratedContent.stepExplanation?.map((step, i) => (
                              <p key={i} className="text-xs sm:text-sm text-[#3D352E] bg-white p-3 rounded-xl border border-[#DCE8E0] leading-relaxed">
                                {step}
                              </p>
                            ))}
                          </div>

                          <div className="flex items-center justify-between pt-3">
                            <div className="flex items-center gap-2 text-xs font-bold text-[#5C5248]">
                              <MochiKumaIcon size={32} />
                              <span>{aiGeneratedContent.coachComment}</span>
                            </div>
                            <button
                              type="button"
                              onClick={() => {
                                if (!solvedQuestionIds.includes(activeQuestion.id)) {
                                  setSolvedQuestionIds(prev => [...prev, activeQuestion.id]);
                                }
                                setStudyLogs(prev => [
                                  { id: `log-${Date.now()}`, date: new Date().toLocaleDateString('ja-JP'), title: `AI道場クリア：${activeQuestion.subject}「${activeQuestion.unit}」` },
                                  ...prev
                                ]);
                                showToast(`「${activeQuestion.unit}」の演習をクリアしました！`, 20);
                                setActiveQuestion(null);
                              }}
                              className="bg-[#4B8364] hover:bg-[#3B6E52] text-white text-xs sm:text-sm font-extrabold px-5 py-2.5 rounded-xl shadow-sm transition flex items-center gap-1.5"
                            >
                              <CheckCircle2 className="w-4 h-4" />
                              理解できた！クリアにする（+20🌰）
                            </button>
                          </div>
                        </div>
                      )}
                    </div>
                  )
                )}
              </div>
            )}

            {/* 登録問題からの「間違い復習（基準正答率以上）」＆「得意の応用」リスト */}
            <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
              {/* 左：間違えた問題（正答率基準フィルター連動） */}
              <div className="bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-5 sm:p-6 shadow-sm space-y-4">
                <div className="flex items-center justify-between">
                  <div>
                    <span className="bg-[#F5E6E4] text-[#A64B44] text-xs font-extrabold px-3 py-1 rounded-full">
                      復習優先（正答率 {accuracyThreshold}%以上 × まちがえた問題）
                    </span>
                    <h3 className="text-base font-extrabold text-[#3D352E] mt-2">
                      詳しい解説と類題で解き直し
                    </h3>
                  </div>
                </div>

                {allQuestions.filter(q => !q.isCorrect && q.accuracyRate >= accuracyThreshold).length === 0 ? (
                  <div className="bg-[#F9F6F0] border border-dashed border-[#D8C5A5] rounded-2xl p-6 text-center space-y-2">
                    <p className="text-xs sm:text-sm font-bold text-[#5C5248]">
                      {allQuestions.length === 0
                        ? 'まだテスト問題が登録されていません。テストを追加するか、下のサンプル演習をお試しください。'
                        : `正答率${accuracyThreshold}%以上で間違えた問題はありません！基準を「30%以上」などに下げてみることもできます。`}
                    </p>
                    <button
                      onClick={() =>
                        startDojoExercise(
                          {
                            id: 'sample-review',
                            subject: '算数',
                            number: 'サンプル演習[8](2)',
                            unit: '和差算',
                            isCorrect: false,
                            accuracyRate: 55,
                            questionSummary: 'クッキーを2人で分けてお金をやり取りする和差算'
                          },
                          'review'
                        )
                      }
                      className="mt-2 bg-white hover:bg-[#F3EFE6] text-[#4B8364] border border-[#4B8364] text-xs font-bold px-4 py-2 rounded-full transition"
                    >
                      ✏️ サンプル問題（和差算の解き直し）を体験する
                    </button>
                  </div>
                ) : (
                  <div className="space-y-3">
                    {allQuestions
                      .filter(q => !q.isCorrect && q.accuracyRate >= accuracyThreshold)
                      .map(q => (
                        <div key={q.id} className="p-4 rounded-2xl bg-[#FAF8F4] border border-[#E2D9C8] flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                          <div className="space-y-1">
                            <div className="flex items-center gap-2">
                              <span className={`text-[11px] font-bold px-2 py-0.5 rounded-md ${SUBJECT_STYLES[q.subject]?.bg} ${SUBJECT_STYLES[q.subject]?.text}`}>
                                {q.subject}
                              </span>
                              <span className="font-extrabold text-sm text-[#3D352E]">{q.unit}</span>
                              <span className="text-xs font-bold text-[#A64B44] bg-[#F5E6E4] px-2 py-0.5 rounded-full">
                                正答率 {q.accuracyRate}%
                              </span>
                              {q.isSolved && (
                                <span className="text-xs font-bold text-[#4B8364] bg-[#E6EFE9] px-2 py-0.5 rounded-full">
                                  ✓ 解き直し済
                                </span>
                              )}
                            </div>
                            <p className="text-xs text-[#5C5248]">{q.questionSummary}</p>
                            <p className="text-[10px] text-[#7A6F63]">{q.testName} / {q.number}</p>
                          </div>
                          <button
                            onClick={() => startDojoExercise(q, 'review')}
                            className="bg-[#4B8364] hover:bg-[#3B6E52] text-white text-xs font-bold px-4 py-2 rounded-xl shrink-0 transition"
                          >
                            解説＆類題をとく
                          </button>
                        </div>
                      ))}
                  </div>
                )}
              </div>

              {/* 右：得意な問題（正解できた問題 → 応用問題） */}
              <div className="bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-5 sm:p-6 shadow-sm space-y-4">
                <div>
                  <span className="bg-[#E6EFE9] text-[#3B6E52] text-xs font-extrabold px-3 py-1 rounded-full">
                    得意を伸ばす（テストで正解できた単元）
                  </span>
                  <h3 className="text-base font-extrabold text-[#3D352E] mt-2">
                    発展・応用問題にチャレンジ
                  </h3>
                </div>

                {allQuestions.filter(q => q.isCorrect).length === 0 ? (
                  <div className="bg-[#F9F6F0] border border-dashed border-[#D8C5A5] rounded-2xl p-6 text-center space-y-2">
                    <p className="text-xs sm:text-sm font-bold text-[#5C5248]">
                      テストで正解できた得意問題を登録すると、ここから発展的な応用問題に挑戦できます。
                    </p>
                    <button
                      onClick={() =>
                        startDojoExercise(
                          {
                            id: 'sample-adv',
                            subject: '算数',
                            number: 'サンプル応用',
                            unit: '分配算・和差算',
                            isCorrect: true,
                            accuracyRate: 68,
                            questionSummary: '3人でノートを分ける応用問題'
                          },
                          'advanced'
                        )
                      }
                      className="mt-2 bg-white hover:bg-[#F3EFE6] text-[#4B8364] border border-[#4B8364] text-xs font-bold px-4 py-2 rounded-full transition"
                    >
                      👑 サンプル応用問題を体験する
                    </button>
                  </div>
                ) : (
                  <div className="space-y-3">
                    {allQuestions
                      .filter(q => q.isCorrect)
                      .map(q => (
                        <div key={q.id} className="p-4 rounded-2xl bg-[#FAF8F4] border border-[#E2D9C8] flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                          <div className="space-y-1">
                            <div className="flex items-center gap-2">
                              <span className={`text-[11px] font-bold px-2 py-0.5 rounded-md ${SUBJECT_STYLES[q.subject]?.bg} ${SUBJECT_STYLES[q.subject]?.text}`}>
                                {q.subject}
                              </span>
                              <span className="font-extrabold text-sm text-[#3D352E]">{q.unit}</span>
                              <span className="text-xs font-bold text-[#3B6E52] bg-[#E6EFE9] px-2 py-0.5 rounded-full">
                                正答率 {q.accuracyRate}% ⭕
                              </span>
                            </div>
                            <p className="text-xs text-[#5C5248]">{q.questionSummary}</p>
                            <p className="text-[10px] text-[#7A6F63]">{q.testName} / {q.number}</p>
                          </div>
                          <button
                            onClick={() => startDojoExercise(q, 'advanced')}
                            className="bg-[#547AA5] hover:bg-[#3E5F8A] text-white text-xs font-bold px-4 py-2 rounded-xl shrink-0 transition"
                          >
                            応用問題をだす
                          </button>
                        </div>
                      ))}
                  </div>
                )}
              </div>
            </div>
          </div>
        )}

        {/* ==================== 4. テスト結果・PDF追加画面（iPad対応） ==================== */}
        {activeView === 'upload' && (
          <form onSubmit={handleSaveTest} className="bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-5 sm:p-7 shadow-sm space-y-6">
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 border-b border-[#EFEAE0] pb-4">
              <div>
                <h2 className="text-xl font-extrabold text-[#3D352E] flex items-center gap-2">
                  <Upload className="w-6 h-6 text-[#4B8364]" />
                  テスト結果・答案PDF・解答解説の追加
                </h2>
                <p className="text-xs sm:text-sm text-[#7A6F63] mt-1">
                  iPadから答案や正答率表のPDF・写真を添付し、点数と注目問題（間違えた問題や得意問題）を記録しましょう。
                </p>
              </div>
              <button
                type="button"
                onClick={handleAiScanFiles}
                disabled={isScanningFiles}
                className="bg-[#E6EFE9] hover:bg-[#D5E6DC] text-[#2C5840] border border-[#4B8364] font-extrabold text-xs sm:text-sm px-4 py-2.5 rounded-2xl flex items-center gap-2 shrink-0 transition"
              >
                {isScanningFiles ? (
                  <>
                    <RefreshCw className="w-4 h-4 animate-spin" />
                    AIで読み取り中...
                  </>
                ) : (
                  <>
                    <Sparkles className="w-4 h-4" />
                    AIで書類から問題リストを自動作成
                  </>
                )}
              </button>
            </div>

            {/* PDF・写真添付エリア */}
            <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
              {[
                { key: 'answerSheet', title: '① 答案用紙 (PDF / 写真)', desc: '子どもの答案をiPadで撮影または選択' },
                { key: 'explanation', title: '② 解答・解説 (PDF / 写真)', desc: '解説冊子のPDFや写真' },
                { key: 'accuracySheet', title: '③ 問題ごとの正答率表', desc: 'マイページ等の正答率データ' }
              ].map(item => {
                const f = uploadForm.files[item.key];
                return (
                  <label
                    key={item.key}
                    className={`cursor-pointer rounded-2xl border-2 border-dashed p-4 text-center flex flex-col items-center justify-center transition ${
                      f ? 'bg-[#E6EFE9]/60 border-[#4B8364]' : 'bg-[#F9F6F0] border-[#D8C5A5] hover:bg-[#F3EFE6]'
                    }`}
                  >
                    <input
                      type="file"
                      accept="application/pdf,image/*"
                      onChange={e => handleFileSelect(item.key, e)}
                      className="hidden"
                    />
                    <FileText className="w-6 h-6 text-[#4B8364] mb-1.5" />
                    <p className="font-extrabold text-xs sm:text-sm text-[#3D352E]">{item.title}</p>
                    <p className="text-[11px] text-[#7A6F63] mt-0.5">{item.desc}</p>
                    {f ? (
                      <span className="mt-2 bg-white text-[#2C5840] border border-[#4B8364] text-xs font-bold px-3 py-0.5 rounded-full">
                        ✓ {f.name}
                      </span>
                    ) : (
                      <span className="mt-2 bg-white text-[#5C5248] border border-[#D8C5A5] text-xs font-bold px-3 py-0.5 rounded-full">
                        ファイル・写真を選択
                      </span>
                    )}
                  </label>

              })}
            </div>

            {/* テスト基本情報 */}
            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
              <div>
                <label className="block text-xs font-bold text-[#5C5248] mb-1">テストの種類</label>
                <select
                  value={uploadForm.type}
                  onChange={e => handleTestTypeChange(e.target.value)}
                  className="w-full rounded-xl border border-[#D8C5A5] bg-white px-3 py-2.5 text-sm font-bold text-[#3D352E]"
                >
                  <option value="組分けテスト">四谷大塚 公開組分けテスト</option>
                  <option value="カリキュラムテスト">カリキュラムテスト（コース別）</option>
                  <option value="全統小">全国統一小学生テスト</option>
                </select>
              </div>
              <div>
                <label className="block text-xs font-bold text-[#5C5248] mb-1">テスト名（回数など）</label>
                <input
                  type="text"
                  placeholder="例：第4回 公開組分けテスト"
                  value={uploadForm.name}
                  onChange={e => setUploadForm({ ...uploadForm, name: e.target.value })}
                  className="w-full rounded-xl border border-[#D8C5A5] bg-white px-3 py-2.5 text-sm font-bold"
                />
              </div>
              <div>
                <label className="block text-xs font-bold text-[#5C5248] mb-1">クラス / コース</label>
                <input
                  type="text"
                  placeholder="例：Bコース / Cコース"
                  value={uploadForm.course}
                  onChange={e => setUploadForm({ ...uploadForm, course: e.target.value })}
                  className="w-full rounded-xl border border-[#D8C5A5] bg-white px-3 py-2.5 text-sm font-bold"
                />
              </div>
              <div>
                <label className="block text-xs font-bold text-[#5C5248] mb-1">実施日</label>
                <input
                  type="date"
                  value={uploadForm.date}
                  onChange={e => setUploadForm({ ...uploadForm, date: e.target.value })}
                  className="w-full rounded-xl border border-[#D8C5A5] bg-white px-3 py-2.5 text-sm font-bold"
                />
              </div>
            </div>

            {/* 4科目＋四科合計の得点と偏差値 */}
            <div>
              <h3 className="text-sm font-extrabold text-[#3D352E] mb-2.5">科目ごとの得点と偏差値</h3>
              <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
                {['算数', '国語', '理科', '社会', '四科'].map(sub => (
                  <div key={sub} className={`p-3 rounded-2xl border ${SUBJECT_STYLES[sub].bg} ${SUBJECT_STYLES[sub].border}`}>
                    <div className="flex items-center justify-between mb-2">
                      <span className={`font-extrabold text-sm ${SUBJECT_STYLES[sub].text}`}>{sub}</span>
                      <span className="text-[11px] text-[#7A6F63]">/{uploadForm.scores[sub].maxScore}点</span>
                    </div>
                    <div className="space-y-2">
                      <input
                        type="number"
                        placeholder="得点"
                        value={uploadForm.scores[sub].score}
                        onChange={e =>
                          setUploadForm({
                            ...uploadForm,
                            scores: {
                              ...uploadForm.scores,
                              [sub]: { ...uploadForm.scores[sub], score: e.target.value }
                            }
                          })
                        }
                        className="w-full bg-white rounded-lg px-2.5 py-1.5 text-xs font-bold border border-[#D8C5A5]"
                      />
                      <input
                        type="number"
                        step="0.1"
                        placeholder="偏差値"
                        value={uploadForm.scores[sub].deviation}
                        onChange={e =>
                          setUploadForm({
                            ...uploadForm,
                            scores: {
                              ...uploadForm.scores,
                              [sub]: { ...uploadForm.scores[sub], deviation: e.target.value }
                            }
                          })
                        }
                        className="w-full bg-white rounded-lg px-2.5 py-1.5 text-xs font-bold border border-[#D8C5A5]"
                      />
                    </div>
                  </div>
                ))}
              </div>
            </div>

            {/* 問題ごとの正答率と正誤入力 */}
            <div className="space-y-3">
              <div className="flex items-center justify-between">
                <div>
                  <h3 className="text-sm font-extrabold text-[#3D352E]">
                    問題ごとの正答率と正誤（間違えた問題・伸ばしたい得意問題）
                  </h3>
                  <p className="text-xs text-[#7A6F63]">
                    全体の正答率（%）を入れておくと、30%・40%・50%の基準ごとに復習リストを自動で切り替えられます。
                  </p>
                </div>
                <button
                  type="button"
                  onClick={() =>
                    setUploadForm(prev => ({
                      ...prev,
                      questions: [
                        ...prev.questions,
                        {
                          id: `q-row-${Date.now()}`,
                          subject: '算数',
                          number: `大問${prev.questions.length + 2}`,
                          unit: '',
                          isCorrect: false,
                          accuracyRate: 50,
                          questionSummary: ''
                        }
                      ]
                    }))
                  }
                  className="bg-[#E6EFE9] hover:bg-[#D5E6DC] text-[#2C5840] font-bold text-xs px-3.5 py-2 rounded-xl flex items-center gap-1"
                >
                  <Plus className="w-4 h-4" />
                  問題を追加
                </button>
              </div>

              <div className="space-y-2.5">
                {uploadForm.questions.map((q, idx) => (
                  <div key={q.id} className="p-3 rounded-2xl bg-[#F9F6F0] border border-[#E2D9C8] grid grid-cols-1 md:grid-cols-12 gap-2.5 items-center">
                    <div className="md:col-span-2">
                      <select
                        value={q.subject}
                        onChange={e => {
                          const next = [...uploadForm.questions];
                          next[idx].subject = e.target.value;
                          setUploadForm({ ...uploadForm, questions: next });
                        }}
                        className="w-full rounded-lg border border-[#D8C5A5] bg-white px-2 py-1.5 text-xs font-bold"
                      >
                        <option value="算数">算数</option>
                        <option value="国語">国語</option>
                        <option value="理科">理科</option>
                        <option value="社会">社会</option>
                      </select>
                    </div>
                    <div className="md:col-span-2">
                      <input
                        type="text"
                        placeholder="問題番号 (例: 大問3(2))"
                        value={q.number}
                        onChange={e => {
                          const next = [...uploadForm.questions];
                          next[idx].number = e.target.value;
                          setUploadForm({ ...uploadForm, questions: next });
                        }}
                        className="w-full rounded-lg border border-[#D8C5A5] bg-white px-2.5 py-1.5 text-xs font-bold"
                      />
                    </div>
                    <div className="md:col-span-2">
                      <input
                        type="text"
                        placeholder="単元名 (例: 和差算)"
                        value={q.unit}
                        onChange={e => {
                          const next = [...uploadForm.questions];
                          next[idx].unit = e.target.value;
                          setUploadForm({ ...uploadForm, questions: next });
                        }}
                        className="w-full rounded-lg border border-[#D8C5A5] bg-white px-2.5 py-1.5 text-xs font-bold"
                      />
                    </div>
                    <div className="md:col-span-3">
                      <input
                        type="text"
                        placeholder="問題の内容メモ"
                        value={q.questionSummary}
                        onChange={e => {
                          const next = [...uploadForm.questions];
                          next[idx].questionSummary = e.target.value;
                          setUploadForm({ ...uploadForm, questions: next });
                        }}
                        className="w-full rounded-lg border border-[#D8C5A5] bg-white px-2.5 py-1.5 text-xs"
                      />
                    </div>
                    <div className="md:col-span-2 flex items-center gap-1">
                      <span className="text-[11px] font-bold text-[#7A6F63] shrink-0">正答率</span>
                      <input
                        type="number"
                        min="0"
                        max="100"
                        value={q.accuracyRate}
                        onChange={e => {
                          const next = [...uploadForm.questions];
                          next[idx].accuracyRate = e.target.value;
                          setUploadForm({ ...uploadForm, questions: next });
                        }}
                        className="w-16 rounded-lg border border-[#D8C5A5] bg-white px-2 py-1.5 text-xs font-bold"
                      />
                      <span className="text-xs text-[#7A6F63]">%</span>
                    </div>
                    <div className="md:col-span-1 flex items-center justify-end gap-1">
                      <button
                        type="button"
                        onClick={() => {
                          const next = [...uploadForm.questions];
                          next[idx].isCorrect = !next[idx].isCorrect;
                          setUploadForm({ ...uploadForm, questions: next });
                        }}
                        className={`px-2.5 py-1.5 rounded-lg text-xs font-extrabold ${
                          q.isCorrect ? 'bg-[#E6EFE9] text-[#2C5840]' : 'bg-[#F5E6E4] text-[#A64B44]'
                        }`}
                      >
                        {q.isCorrect ? '⭕正解' : '❌不正解'}
                      </button>
                    </div>
                  </div>
                ))}
              </div>
            </div>

            <div className="flex justify-end gap-3 pt-2">
              <button
                type="button"
                onClick={() => setActiveView('home')}
                className="bg-[#F3EFE6] hover:bg-[#E5DEC9] text-[#5C5248] font-bold px-5 py-3 rounded-full text-sm transition"
              >
                キャンセル
              </button>
              <button
                type="submit"
                className="bg-[#4B8364] hover:bg-[#3B6E52] text-white font-extrabold px-8 py-3 rounded-full shadow-sm text-sm transition"
              >
                このテスト結果を登録する（+30🌰）
              </button>
            </div>
          </form>
        )}

        {/* ==================== 5. 暗記カード（理科・社会・国語の一問一答） ==================== */}
        {activeView === 'flashcards' && (
          <div className="space-y-6">
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
              <div>
                <h2 className="text-2xl font-extrabold text-[#3D352E] flex items-center gap-2">
                  <span>🎴</span>
                  暗記カード（一問一答・自分で作成）
                </h2>
                <p className="text-xs sm:text-sm text-[#7A6F63] mt-0.5">
                  最初はデータが空です。覚えたい理科・社会の用語や、テストで間違えた問題から自分だけの暗記カードを作れます！
                </p>
              </div>

              {allQuestions.filter(q => !q.isCorrect).length > 0 && (
                <button
                  type="button"
                  onClick={() => {
                    const wrongQs = allQuestions.filter(q => !q.isCorrect);
                    const created = wrongQs.map((q, i) => ({
                      id: `fc-auto-${Date.now()}-${i}`,
                      subject: q.subject,
                      unit: q.unit,
                      front: `【${q.unit}】${q.questionSummary}`,
                      back: `テスト（${q.testName} ${q.number}）の見直しポイントを確認しよう！`,
                      mastered: false
                    }));
                    setFlashcards(prev => [...prev, ...created]);
                    showToast(`テストの間違い問題から ${created.length}枚のカードを作成しました！`, 15);
                  }}
                  className="bg-[#E6EFE9] hover:bg-[#D5E6DC] text-[#2C5840] border border-[#4B8364] text-xs font-extrabold px-4 py-2.5 rounded-full transition"
                >
                  ＋ テストの間違い問題から自動作成する
                </button>
              )}
            </div>

            <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
              {/* 左7列：カードめくり演習ゾーン */}
              <div className="lg:col-span-7 bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-5 sm:p-6 shadow-sm flex flex-col justify-between space-y-5">
                <div className="flex items-center justify-between">
                  <div className="flex gap-1.5">
                    {['すべて', '理科', '社会', '算数', '国語'].map(sub => (
                      <button
                        key={sub}
                        onClick={() => {
                          setCardSubjectFilter(sub);
                          setCurrentCardIndex(0);
                          setIsCardFlipped(false);
                        }}
                        className={`px-3 py-1 rounded-full text-xs font-bold transition ${
                          cardSubjectFilter === sub
                            ? 'bg-[#4B8364] text-white'
                            : 'bg-[#F3EFE6] text-[#7A6F63]'
                        }`}
                      >
                        {sub}
                      </button>
                    ))}
                  </div>
                  <span className="text-xs font-bold text-[#7A6F63]">
                    {filteredCards.length > 0 ? `${currentCardIndex + 1} / ${filteredCards.length} 枚目` : '0 枚'}
                  </span>
                </div>

                {filteredCards.length === 0 ? (
                  <div className="bg-[#F9F6F0] border border-dashed border-[#D8C5A5] rounded-2xl h-64 p-6 flex flex-col items-center justify-center text-center space-y-3">
                    <MochiKumaIcon size={52} />
                    <p className="text-sm font-bold text-[#5C5248]">
                      まだ暗記カードがありません。右のフォームから最初の1枚を作ってみよう！
                    </p>
                  </div>
                ) : (
                  (() => {
                    const card = filteredCards[currentCardIndex] || filteredCards[0];
                    return (
                      <div className="space-y-4">
                        <div
                          onClick={() => setIsCardFlipped(!isCardFlipped)}
                          className={`cursor-pointer rounded-3xl border-2 p-8 h-64 flex flex-col items-center justify-center text-center transition-all shadow-sm select-none ${
                            isCardFlipped
                              ? 'bg-[#E6EFE9] border-[#4B8364]'
                              : 'bg-[#FAF8F4] border-[#D8C5A5] hover:border-[#4B8364]'
                          }`}
                        >
                          <div className="flex items-center gap-2 mb-3">
                            <span className="text-xs font-extrabold px-2.5 py-0.5 rounded-full bg-white text-[#5C5248] border border-[#D8C5A5]">
                              {card.subject} {card.unit && `・ ${card.unit}`}
                            </span>
                            <span className="text-xs font-bold text-[#7A6F63]">
                              {isCardFlipped ? '💡 うら（こたえ）' : '❓ おもて（もんだい）'}
                            </span>
                          </div>
                          <p className="text-base sm:text-xl font-extrabold text-[#3D352E] leading-relaxed">
                            {isCardFlipped ? card.back : card.front}
                          </p>
                          <p className="text-xs text-[#7A6F63] mt-6">
                            （カードをタップすると裏返ります）
                          </p>
                        </div>

                        <div className="flex items-center justify-between gap-2">
                          <button
                            type="button"
                            onClick={() => {
                              setIsCardFlipped(false);
                              setCurrentCardIndex(prev => (prev > 0 ? prev - 1 : filteredCards.length - 1));
                            }}
                            className="bg-[#F3EFE6] hover:bg-[#E5DEC9] text-[#5C5248] font-bold px-4 py-2.5 rounded-xl text-xs flex items-center gap-1"
                          >
                            <ChevronLeft className="w-4 h-4" /> まえへ
                          </button>

                          <button
                            type="button"
                            onClick={() => {
                              showToast('バッチリ覚えたね！どんぐり+5個', 5);
                              setIsCardFlipped(false);
                              setCurrentCardIndex(prev => (prev < filteredCards.length - 1 ? prev + 1 : 0));
                            }}
                            className="bg-[#4B8364] hover:bg-[#3B6E52] text-white font-extrabold px-5 py-2.5 rounded-xl text-xs"
                          >
                            ⭕ おぼえた！（+5🌰）
                          </button>

                          <button
                            type="button"
                            onClick={() => {
                              setIsCardFlipped(false);
                              setCurrentCardIndex(prev => (prev < filteredCards.length - 1 ? prev + 1 : 0));
                            }}
                            className="bg-[#F3EFE6] hover:bg-[#E5DEC9] text-[#5C5248] font-bold px-4 py-2.5 rounded-xl text-xs flex items-center gap-1"
                          >
                            つぎへ <ChevronRight className="w-4 h-4" />
                          </button>
                        </div>
                      </div>
                    );
                  })()
                )}
              </div>

              {/* 右5列：新しい暗記カードを追加するフォーム */}
              <div className="lg:col-span-5 bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-5 sm:p-6 shadow-sm space-y-4">
                <h3 className="text-base font-extrabold text-[#3D352E]">
                  ＋ 新しい暗記カードを作る
                </h3>
                <div className="grid grid-cols-2 gap-3">
                  <div>
                    <label className="block text-xs font-bold text-[#5C5248] mb-1">科目</label>
                    <select
                      value={newCardForm.subject}
                      onChange={e => setNewCardForm({ ...newCardForm, subject: e.target.value })}
                      className="w-full rounded-xl border border-[#D8C5A5] bg-white px-3 py-2 text-xs font-bold"
                    >
                      <option value="理科">理科</option>
                      <option value="社会">社会</option>
                      <option value="国語">国語（漢字・語句）</option>
                      <option value="算数">算数（公式・単位）</option>
                    </select>
                  </div>
                  <div>
                    <label className="block text-xs font-bold text-[#5C5248] mb-1">単元名</label>
                    <input
                      type="text"
                      placeholder="例：春の生物 / 都道府県"
                      value={newCardForm.unit}
                      onChange={e => setNewCardForm({ ...newCardForm, unit: e.target.value })}
                      className="w-full rounded-xl border border-[#D8C5A5] bg-white px-3 py-2 text-xs font-bold"
                    />
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-bold text-[#5C5248] mb-1">おもて（問題）</label>
                  <textarea
                    rows={2}
                    placeholder="例：モンシロチョウの幼虫が食べる葉っぱ（食草）は何？"
                    value={newCardForm.front}
                    onChange={e => setNewCardForm({ ...newCardForm, front: e.target.value })}
                    className="w-full rounded-xl border border-[#D8C5A5] bg-white p-3 text-xs font-bold"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold text-[#5C5248] mb-1">うら（答え・解説）</label>
                  <textarea
                    rows={2}
                    placeholder="例：キャベツやアブラナなどのアブラナ科の植物"
                    value={newCardForm.back}
                    onChange={e => setNewCardForm({ ...newCardForm, back: e.target.value })}
                    className="w-full rounded-xl border border-[#D8C5A5] bg-white p-3 text-xs font-bold"
                  />
                </div>

                <button
                  type="button"
                  onClick={() => {
                    if (!newCardForm.front.trim() || !newCardForm.back.trim()) {
                      showToast('問題と答えを入力してください', 0);
                      return;
                    }
                    setFlashcards(prev => [
                      ...prev,
                      {
                        id: `fc-${Date.now()}`,
                        subject: newCardForm.subject,
                        unit: newCardForm.unit.trim(),
                        front: newCardForm.front.trim(),
                        back: newCardForm.back.trim()
                      }
                    ]);
                    setNewCardForm({ ...newCardForm, front: '', back: '' });
                    showToast('暗記カードを追加しました！（+10🌰）', 10);
                  }}
                  className="w-full bg-[#4B8364] hover:bg-[#3B6E52] text-white font-extrabold py-3 rounded-full text-xs sm:text-sm transition"
                >
                  カードを追加する（+10🌰）
                </button>
              </div>
            </div>
          </div>
        )}

        {/* ==================== 6. 志望校の設定（最大5校・最初は空欄） ==================== */}
        {activeView === 'schools' && (
          <div className="bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-5 sm:p-7 shadow-sm space-y-6">
            <div>
              <h2 className="text-xl font-extrabold text-[#3D352E] flex items-center gap-2">
                <span>🏫</span>
                志望校の設定（最大5校）
              </h2>
              <p className="text-xs sm:text-sm text-[#7A6F63] mt-1">
                お子様と一緒に目標にしたい中学校名と合格目標偏差値を入力してください。入力した学校のみグラフに目標ラインとして表示されます。
              </p>
            </div>

            <div className="space-y-3">
              {schools.map((school, idx) => (
                <div key={school.id} className="p-4 rounded-2xl bg-[#F9F6F0] border border-[#E2D9C8] flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                  <div className="flex items-center gap-3 flex-1">
                    <span
                      className="w-7 h-7 rounded-full text-white font-extrabold text-xs flex items-center justify-center shrink-0"
                      style={{ backgroundColor: school.color }}
                    >
                      {idx + 1}
                    </span>
                    <input
                      type="text"
                      placeholder={`第${idx + 1}志望の学校名を入力（例：〇〇中学校）`}
                      value={school.name}
                      onChange={e => {
                        const next = [...schools];
                        next[idx].name = e.target.value;
                        setSchools(next);
                      }}
                      className="w-full bg-white rounded-xl border border-[#D8C5A5] px-3.5 py-2 text-sm font-bold text-[#3D352E]"
                    />
                  </div>
                  <div className="flex items-center gap-2">
                    <span className="text-xs font-bold text-[#5C5248]">目標偏差値:</span>
                    <input
                      type="number"
                      min="30"
                      max="80"
                      placeholder="例: 60"
                      value={school.targetDeviation}
                      onChange={e => {
                        const next = [...schools];
                        next[idx].targetDeviation = e.target.value;
                        setSchools(next);
                      }}
                      className="w-24 bg-white rounded-xl border border-[#D8C5A5] px-3 py-2 text-sm font-extrabold text-center text-[#4B8364]"
                    />
                  </div>
                </div>
              ))}
            </div>

            <div className="flex justify-end">
              <button
                type="button"
                onClick={() => {
                  showToast('志望校の設定を保存しました！', 5);
                  setActiveView('home');
                }}
                className="bg-[#4B8364] hover:bg-[#3B6E52] text-white font-extrabold px-6 py-3 rounded-full text-sm transition"
              >
                保存してホームへもどる
              </button>
            </div>
          </div>
        )}

        {/* ==================== 7. おたのしみ（学習のきろく・控えめな育成要素） ==================== */}
        {activeView === 'records' && (
          <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
            <div className="bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-6 shadow-sm flex flex-col items-center text-center space-y-4">
              <span className="bg-[#E6EFE9] text-[#2C5840] text-xs font-extrabold px-3.5 py-1 rounded-full">
                2年間いっしょに歩むパーソナルコーチ
              </span>
              <div className="w-32 h-32 rounded-full bg-[#F9F6F0] border-2 border-[#D8C5A5] flex items-center justify-center shadow-inner">
                <MochiKumaIcon size={88} />
              </div>
              <div>
                <p className="text-xs font-bold text-[#7A6F63]">レベル {coachStage.level}</p>
                <h3 className="text-xl font-extrabold text-[#3D352E] mt-0.5">
                  {coachStage.badge} {coachStage.title}
                </h3>
                <p className="text-xs sm:text-sm text-[#5C5248] mt-2">{coachStage.desc}</p>
              </div>

              <div className="w-full bg-[#F9F6F0] p-4 rounded-2xl border border-[#E2D9C8] space-y-2">
                <div className="flex justify-between text-xs font-bold text-[#5C5248]">
                  <span>集めたどんぐり</span>
                  <span>🌰 {acorns} / {coachStage.next} 個</span>
                </div>
                <div className="w-full h-3 bg-[#E5DEC9] rounded-full overflow-hidden">
                  <div
                    className="h-full bg-[#4B8364] rounded-full transition-all"
                    style={{ width: `${Math.min(100, Math.round((acorns / coachStage.next) * 100))}%` }}
                  />
                </div>
              </div>
            </div>

            <div className="bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-6 shadow-sm space-y-4">
              <h3 className="text-base font-extrabold text-[#3D352E] flex items-center gap-2">
                <Calendar className="w-5 h-5 text-[#4B8364]" />
                これまでの学習のきろく
              </h3>
              {studyLogs.length === 0 ? (
                <div className="bg-[#F9F6F0] border border-dashed border-[#D8C5A5] rounded-2xl p-6 text-center text-xs sm:text-sm text-[#7A6F63] font-bold">
                  まだ学習のきろくがありません。テスト登録やAI道場の解き直しをすると、ここに足あとが残ります！
                </div>
              ) : (
                <div className="space-y-2 max-h-80 overflow-y-auto">
                  {studyLogs.map(log => (
                    <div key={log.id} className="p-3 rounded-xl bg-[#FAF8F4] border border-[#EAE3D5] flex items-center justify-between text-xs">
                      <span className="font-bold text-[#3D352E]">{log.title}</span>
                      <span className="text-[#7A6F63]">{log.date}</span>
                    </div>
                  ))}
                </div>
              )}
            </div>
          </div>
        )}

        {/* ==================== 8. iPad共有・データ引き継ぎ画面 ==================== */}
        {activeView === 'share' && (
          <div className="bg-[#FFFDF9] border border-[#E2D9C8] rounded-3xl p-5 sm:p-7 shadow-sm space-y-6">
            <div>
              <h2 className="text-xl font-extrabold text-[#3D352E] flex items-center gap-2">
                <Share2 className="w-6 h-6 text-[#4B8364]" />
                子どものiPadでの利用・データ共有ガイド
              </h2>
              <p className="text-xs sm:text-sm text-[#7A6F63] mt-1">
                お子様のiPadのSafari等でこのページのURLを開き、「ホーム画面に追加」すると本物のアプリのように使えます。
              </p>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
              {/* URL共有とホーム画面追加手順 */}
              <div className="bg-[#F9F6F0] border border-[#E2D9C8] rounded-2xl p-5 space-y-3">
                <h3 className="text-sm font-extrabold text-[#3D352E]">
                  📱 Step 1：iPadでアプリとして使う方法
                </h3>
                <ol className="text-xs text-[#5C5248] space-y-2 list-decimal list-inside leading-relaxed font-medium">
                  <li>このページのURLをお子様のiPad（メッセージやAirDrop等）に送ります。</li>
                  <li>iPadのSafariで開き、画面上の共有ボタン（四角から上矢印）をタップして<strong>「ホーム画面に追加」</strong>を選ぶと、いつでもワンタップで開けます。</li>
                  <li>入力したデータはブラウザ内に自動保存されます。</li>
                </ol>
                <button
                  type="button"
                  onClick={() => copyToClipboard(window.location.href, '現在のページURLをコピーしました！')}
                  className="bg-[#4B8364] hover:bg-[#3B6E52] text-white font-bold text-xs px-4 py-2.5 rounded-xl flex items-center gap-1.5"
                >
                  <Copy className="w-4 h-4" />
                  このアプリのURLをコピーする
                </button>
              </div>

              {/* 端末間データ引き継ぎコード */}
              <div className="bg-[#F9F6F0] border border-[#E2D9C8] rounded-2xl p-5 space-y-3">
                <h3 className="text-sm font-extrabold text-[#3D352E]">
                  🔄 Step 2：親のスマホ・PC ⇄ 子どものiPadでデータを同期する
                </h3>
                <p className="text-xs text-[#5C5248] leading-relaxed">
                  別の端末に入力したテスト結果や志望校データを移したい時は、「引き継ぎコードをコピー」して、もう片方の端末に貼り付けるだけで一瞬で同期できます。
                </p>
                <div className="flex flex-wrap gap-2">
                  <button
                    type="button"
                    onClick={exportBackupCode}
                    className="bg-white hover:bg-[#EFEAE0] text-[#2C5840] border border-[#4B8364] font-extrabold text-xs px-4 py-2 rounded-xl flex items-center gap-1.5"
                  >
                    <Download className="w-4 h-4" />
                    現在のデータ引き継ぎコードをコピー
                  </button>
                </div>

                <div className="pt-2 space-y-2">
                  <input
                    type="text"
                    placeholder="ここにコピーした引き継ぎコードを貼り付け"
                    value={backupCodeInput}
                    onChange={e => setBackupCodeInput(e.target.value)}
                    className="w-full bg-white rounded-xl border border-[#D8C5A5] px-3 py-2 text-xs"
                  />
                  <button
                    type="button"
                    onClick={importBackupCode}
                    className="bg-[#547AA5] hover:bg-[#3E5F8A] text-white font-bold text-xs px-4 py-2 rounded-xl"
                  >
                    コードを読み込んでデータを同期する
                  </button>
                </div>
              </div>
            </div>

            {/* 合言葉（パスコード）の変更エリア */}
            <div className="bg-[#F9F6F0] border border-[#E2D9C8] rounded-2xl p-5 space-y-3">
              <h3 className="text-sm font-extrabold text-[#3D352E] flex items-center gap-2">
                <KeyRound className="w-4 h-4 text-[#4B8364]" />
                🔒 家族専用の「合言葉（パスコード）」を変更する
              </h3>
              <p className="text-xs text-[#5C5248]">
                現在の合言葉：<strong className="text-[#4B8364]">{passcode}</strong>（変更すると次回起動時から新しい合言葉でロックされます）
              </p>
              <div className="flex flex-col sm:flex-row gap-2 max-w-md">
                <input
                  type="text"
                  placeholder="新しい合言葉・数字を入力（例: 0415）"
                  value={newPasscodeInput}
                  onChange={e => setNewPasscodeInput(e.target.value)}
                  className="flex-1 bg-white rounded-xl border border-[#D8C5A5] px-3 py-2 text-xs font-bold"
                />
                <button
                  type="button"
                  onClick={() => {
                    if (!newPasscodeInput.trim()) {
                      showToast('新しい合言葉を入力してください', 0);
                      return;
                    }
                    setPasscode(newPasscodeInput.trim());
                    setNewPasscodeInput('');
                    showToast('合言葉を更新しました！', 0);
                  }}
                  className="bg-[#4B8364] hover:bg-[#3B6E52] text-white font-bold text-xs px-5 py-2 rounded-xl shrink-0"
                >
                  合言葉を変更する
                </button>
              </div>
            </div>
          </div>
        )}
      </main>
    </div>
  );
}
